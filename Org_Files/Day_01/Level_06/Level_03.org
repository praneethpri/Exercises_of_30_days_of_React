class Statistics {
    constructor (arr) {
        this.arr = arr
    }

    getCount() {
        return this.arr.length
    }

    getSum() {
        return this.arr.reduce((acc, cur) => acc + cur, 0)
    }

    getMin() {
        return Math.min(...this.arr)
    }

    getMax() {
        return Math.max(...this.arr)
    }

    getRange() {
        return this.getMax() - this.getMin()
    }

    getMean() {
        return Math.round(this.getSum() / this.getCount())
    }

    getMedian() {
        let sortedArr = this.arr.sort()
        return sortedArr[Math.floor(this.getCount() / 2)]
    }

    getMode() {
        let numbers = this.arr.map((i) => this.arr.filter((j) => j === i).length)
        let maximumNumber = Math.max(...numbers)
        let index = numbers.indexOf(maximumNumber)
        let returns = { mode: this.arr[maximumNumber], count: maximumNumber }
        return returns
    }

    getVariance() {
        let subtractions = this.arr.map((i) => i - this.getMean())
        let squred = subtractions.map((j) => j * j)
        let sumOfSqured = squred.reduce((acc, cur) => acc + cur, 0)
        return sumOfSqured / this.getCount()
    }

    getStandardDeviation() {
        return Math.sqrt(this.getVariance())
    }

    getFrequencyDistribution() {
        let numbers = this.arr.map((i) => this.arr.filter((j) => j === i))
        let uniques = [...new Set(this.arr)].sort()
        let frequency = uniques.map((i) => numbers.find((j) => j[0] === i))
        let returns = frequency.map((i) => { percentage : (i.length / this.getCount()) * 100 })
        return returns
    }

    }
