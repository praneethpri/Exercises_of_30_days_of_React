1) Find the person who has many skills in the users object.
   #+begin_src js :results output :noweb yes
<<object-exercise>>
let personHasTheMostSkills;

let maxNumberArray = Array();
for (let i in users) {
    maxNumberArray.push(users[i].skills.length)
}

let maxNumber = Math.max(...maxNumberArray)

let personThatHasMostSkills;
for (let j in users) {
    if (users[j].skills.length === maxNumber) {
        personThatHasMostSkills = j
    }
    else {
        continue
    }
}

console.log(personThatHasMostSkills);
